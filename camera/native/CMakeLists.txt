cmake_minimum_required(VERSION 3.24)
project(MaxNativeCamera LANGUAGES CXX)
if(NOT WIN32)
 message(FATAL_ERROR "The native camera requires Windows")
endif()
find_package(Git REQUIRED)
set(sample_root "${CMAKE_CURRENT_BINARY_DIR}/microsoft-samples")
set(sample_commit "434f6002bdf9cf9829406c3ff2b33387982d6168")
if(NOT EXISTS "${sample_root}/pinned-camera-sources.txt")
 function(camera_git)
  execute_process(COMMAND ${GIT_EXECUTABLE} ${ARGN} RESULT_VARIABLE result)
  if(NOT result EQUAL 0)
   message(FATAL_ERROR "Cannot fetch the pinned Microsoft DirectShow source")
  endif()
 endfunction()
 camera_git(clone --depth 1 --filter=blob:none --sparse https://github.com/microsoft/Windows-classic-samples.git "${sample_root}")
 camera_git(-C "${sample_root}" fetch --depth 1 origin ${sample_commit})
 camera_git(-C "${sample_root}" checkout --detach FETCH_HEAD)
 camera_git(-C "${sample_root}" sparse-checkout set Samples/Win7Samples/multimedia/directshow/baseclasses)
 file(WRITE "${sample_root}/pinned-camera-sources.txt" "${sample_commit}")
endif()
set(base "${sample_root}/Samples/Win7Samples/multimedia/directshow/baseclasses")
file(GLOB base_sources "${base}/*.cpp")
add_library(MaxDShowBase STATIC ${base_sources})
target_include_directories(MaxDShowBase PUBLIC "${base}")
target_compile_definitions(MaxDShowBase PUBLIC UNICODE _UNICODE)
set_property(TARGET MaxDShowBase PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(MaxNativeCamera SHARED NativeCamera.cpp NativeCamera.def)
target_compile_features(MaxNativeCamera PRIVATE cxx_std_17)
set_target_properties(MaxNativeCamera PROPERTIES OUTPUT_NAME "PNGTuberMax.Camera" MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_link_libraries(MaxNativeCamera PRIVATE MaxDShowBase strmiids winmm ole32 oleaut32 uuid shell32 advapi32)
add_executable(MaxNativeNegotiationTests ../../tests/CameraNegotiationTests.cpp)
target_link_libraries(MaxNativeNegotiationTests PRIVATE ole32 oleaut32 strmiids)
